cmake_minimum_required(VERSION 3.20)

# Обычный ESP-IDF-проект: не обращается к исходникам или CMake ядра LexxOS.
set(IDF_TARGET "esp32" CACHE STRING "Target chip")
if (NOT IDF_TARGET STREQUAL "esp32")
    message(FATAL_ERROR "This example targets ESP32")
endif ()
set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig" CACHE FILEPATH "Example configuration")
set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults")
set(COMPONENTS main)
set(LEXXOS_CE_EXECUTION "cooperative" CACHE STRING "CE execution: cooperative or freertos")
set_property(CACHE LEXXOS_CE_EXECUTION PROPERTY STRINGS cooperative freertos)
if (NOT LEXXOS_CE_EXECUTION MATCHES "^(cooperative|freertos)$")
    message(FATAL_ERROR "LEXXOS_CE_EXECUTION must be cooperative or freertos")
endif ()
set(LEXXOS_CE_SDK "" CACHE PATH "Prebuilt CE SDK directory (include/ and lib/)")
set(LEXXOS_CE_LIBRARY "" CACHE FILEPATH "Optional explicit path to liblexxos_ce.a")
set(LEXXOS_CE_INCLUDE_DIR "" CACHE PATH "Optional explicit path to CE include directory")
if (NOT LEXXOS_CE_LIBRARY AND LEXXOS_CE_SDK)
    set(LEXXOS_CE_LIBRARY "${LEXXOS_CE_SDK}/lib/liblexxos_ce.a")
endif ()
if (NOT LEXXOS_CE_INCLUDE_DIR AND LEXXOS_CE_SDK)
    set(LEXXOS_CE_INCLUDE_DIR "${LEXXOS_CE_SDK}/include")
endif ()
if (NOT EXISTS "${LEXXOS_CE_LIBRARY}" OR NOT EXISTS "${LEXXOS_CE_INCLUDE_DIR}/lexxos/ce/build_config.h")
    message(FATAL_ERROR "Provide -DLEXXOS_CE_SDK=/path/to/sdk or both LEXXOS_CE_LIBRARY and LEXXOS_CE_INCLUDE_DIR")
endif ()
get_filename_component(LEXXOS_CE_LIBRARY "${LEXXOS_CE_LIBRARY}" ABSOLUTE)
get_filename_component(LEXXOS_CE_INCLUDE_DIR "${LEXXOS_CE_INCLUDE_DIR}" ABSOLUTE)
# Значения нужны и изолированному requirements-pass ESP-IDF.
set(ENV{ANALOG_PWM_CE_LIBRARY} "${LEXXOS_CE_LIBRARY}")
set(ENV{ANALOG_PWM_CE_INCLUDE_DIR} "${LEXXOS_CE_INCLUDE_DIR}")
set(ENV{ANALOG_PWM_CE_EXECUTION} "${LEXXOS_CE_EXECUTION}")

if (NOT EXISTS "$ENV{IDF_PATH}/tools/cmake/project.cmake")
    message(FATAL_ERROR "Load the ESP-IDF environment (export.sh) before configuring this example")
endif ()
include("$ENV{IDF_PATH}/tools/cmake/project.cmake")
project(LexxCEAnalogPwm)
